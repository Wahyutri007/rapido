import { router, useLocalSearchParams } from "expo-router";
import OrderTypeModifyScreen from "@/components/feature/manage/settings/OrderTypeModifyScreen";
import ManageFormNotFound from "@/components/feature/manage/settings/ManageFormNotFound";
import { ORDER_TYPE_ITEMS } from "@/constants/data/manage/order-type";

export default function OrderTypeModifyScreenRoute() {
	const params = useLocalSearchParams<{ id?: string | string[] }>();
	const id = Array.isArray(params.id) ? params.id[0] : params.id;
	const item = ORDER_TYPE_ITEMS.find((entry) => entry.id === id);
	if (id !== undefined && !item) {
		return (
			<ManageFormNotFound
				entity="Tipe Pesanan"
				onBack={() => router.replace("/manage/order-type")}
			/>
		);
	}
	return (
		<OrderTypeModifyScreen
			key={id === undefined ? "create" : `edit:${id}`}
			item={item}
		/>
	);
}
