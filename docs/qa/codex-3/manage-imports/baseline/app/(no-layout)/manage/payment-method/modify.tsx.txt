import { router, useLocalSearchParams } from "expo-router";
import PaymentMethodModifyScreen from "@/components/feature/manage/settings/PaymentMethodModifyScreen";
import ManageFormNotFound from "@/components/feature/manage/settings/ManageFormNotFound";
import { PAYMENT_METHOD_ITEMS } from "@/constants/data/manage/payment-method";

export default function PaymentMethodModifyScreenRoute() {
	const params = useLocalSearchParams<{ id?: string | string[] }>();
	const id = Array.isArray(params.id) ? params.id[0] : params.id;
	const item = PAYMENT_METHOD_ITEMS.find((entry) => entry.id === id);
	if (id !== undefined && !item) {
		return (
			<ManageFormNotFound
				entity="Metode Pembayaran"
				onBack={() => router.replace("/manage/payment-method")}
			/>
		);
	}
	return (
		<PaymentMethodModifyScreen
			key={id === undefined ? "create" : `edit:${id}`}
			item={item}
		/>
	);
}
